`src/i18n/LangProvider.tsx`:

```tsx
import { createContext, useCallback, useEffect, useMemo, useState, type ReactNode } from 'react';
import type { Lang } from './types';

const STORAGE_KEY = 'lang';

function isLang(v: string | null | undefined): v is Lang {
  return v === 'pt' || v === 'en';
}

/**
 * Ordem de decisão: ?lang= na URL > escolha salva > idioma do navegador.
 * Qualquer valor desconhecido cai no português.
 */
export function resolveInitialLang(search: string, stored: string | null, navigatorLang: string): Lang {
  const fromUrl = new URLSearchParams(search).get('lang');
  if (isLang(fromUrl)) return fromUrl;
  if (isLang(stored)) return stored;
  return navigatorLang.toLowerCase().startsWith('en') ? 'en' : 'pt';
}

function readStored(): string | null {
  try {
    return window.localStorage.getItem(STORAGE_KEY);
  } catch {
    return null;
  }
}

function writeStored(lang: Lang) {
  try {
    window.localStorage.setItem(STORAGE_KEY, lang);
  } catch {
    // Sem armazenamento (aba anônima, bloqueio): o idioma vale só nesta visita.
  }
}

export interface LangContextValue {
  lang: Lang;
  setLang: (lang: Lang) => void;
}

export const LangContext = createContext<LangContextValue | null>(null);

export function LangProvider({ children }: { children: ReactNode }) {
  const [lang, setLangState] = useState<Lang>(() =>
    resolveInitialLang(window.location.search, readStored(), navigator.language ?? 'pt'),
  );

  useEffect(() => {
    document.documentElement.lang = lang === 'pt' ? 'pt-BR' : 'en';
  }, [lang]);

  const setLang = useCallback((next: Lang) => {
    setLangState(next);
    writeStored(next);
    const url = new URL(window.location.href);
    url.searchParams.set('lang', next);
    window.history.replaceState(null, '', url);
  }, []);

  const value = useMemo(() => ({ lang, setLang }), [lang, setLang]);
  return <LangContext.Provider value={value}>{children}</LangContext.Provider>;
}
```

O teste espera `document.documentElement.lang` igual a `'en'` depois da troca; para PT o valor é `'pt-BR'`, que casa com o `index.html`.

`src/i18n/useT.ts`:

```ts
import { useCallback, useContext } from 'react';
import { LangContext } from './LangProvider';
import type { Localized } from './types';

export function useT() {
  const ctx = useContext(LangContext);
  if (!ctx) throw new Error('useT precisa estar dentro de <LangProvider>');
  const { lang, setLang } = ctx;
  const t = useCallback(<T,>(v: Localized<T>): T => v[lang], [lang]);
  return { lang, setLang, t };
}
```

```tsx
import { motion, useReducedMotion } from 'motion/react';

interface BlurTextProps {
  text: string;
  as?: 'h1' | 'h2' | 'h3' | 'p' | 'span';
  className?: string;
  /** Atraso inicial em segundos. */
  delay?: number;
  /** A partir deste índice de palavra, usa o tom secundário (títulos em dois tons). */
  dimFrom?: number;
  /** Índices exatos a escurecer; tem prioridade sobre dimFrom (tons alternados). */
  dimWords?: number[];
  dimClassName?: string;
}

/**
 * Título que sai do desfoque palavra por palavra ao entrar na tela.
 * O texto completo vai em aria-label; as palavras animadas ficam aria-hidden,
 * para o leitor de tela não soletrar pedaço por pedaço.
 */
export default function BlurText({
  text,
  as = 'span',
  className = '',
  delay = 0,
  dimFrom,
  dimWords,
  dimClassName = 'text-muted/60',
}: BlurTextProps) {
  const reduce = useReducedMotion();
  const Tag = as;
  const words = text.split(' ');
  const isDim = (i: number) => (dimWords ? dimWords.includes(i) : dimFrom !== undefined && i >= dimFrom);

  return (
    <Tag aria-label={text} className={className}>
      {words.map((word, i) => (
        <motion.span
          key={`${word}-${i}`}
          data-word
          aria-hidden
          className={`inline-block will-change-[filter,transform] ${isDim(i) ? dimClassName : ''}`}
          initial={reduce ? false : { opacity: 0, filter: 'blur(12px)', y: 20 }}
          whileInView={{ opacity: 1, filter: 'blur(0px)', y: 0 }}
          viewport={{ once: true, margin: '0px 0px -10% 0px' }}
          transition={{ duration: 0.7, delay: delay + i * 0.04, ease: [0.16, 1, 0.3, 1] }}
        >
          {word}
          {i < words.length - 1 ? '\u00a0' : ''}
        </motion.span>
      ))}
    </Tag>
  );
}
```

```tsx
import { motion, useMotionValue, useReducedMotion, useSpring } from 'motion/react';
import type { ReactNode, PointerEvent } from 'react';

/** Puxa levemente o filho na direção do cursor. Só em ponteiro fino, nunca com movimento reduzido. */
export default function Magnetic({ children, strength = 0.25 }: { children: ReactNode; strength?: number }) {
  const reduce = useReducedMotion();
  const x = useSpring(useMotionValue(0), { stiffness: 300, damping: 20, mass: 0.4 });
  const y = useSpring(useMotionValue(0), { stiffness: 300, damping: 20, mass: 0.4 });

  function onMove(e: PointerEvent<HTMLSpanElement>) {
    if (reduce || e.pointerType !== 'mouse') return;
    const r = e.currentTarget.getBoundingClientRect();
    x.set((e.clientX - (r.left + r.width / 2)) * strength);
    y.set((e.clientY - (r.top + r.height / 2)) * strength);
  }

  return (
    <motion.span
      className="inline-block"
      style={{ x, y }}
      onPointerMove={onMove}
      onPointerLeave={() => { x.set(0); y.set(0); }}
    >
      {children}
    </motion.span>
  );
}
```

```tsx
import { animate, useInView, useReducedMotion } from 'motion/react';
import { useEffect, useRef, useState } from 'react';

/** Conta de 0 até value quando entra na tela, uma vez. Com movimento reduzido, mostra o valor direto. */
export default function CountUp({ value, suffix = '', className = '' }: { value: number; suffix?: string; className?: string }) {
  const ref = useRef<HTMLSpanElement>(null);
  const inView = useInView(ref, { once: true, margin: '0px 0px -15% 0px' });
  const reduce = useReducedMotion();
  const [shown, setShown] = useState(reduce ? value : 0);

  useEffect(() => {
    if (!inView) return;
    if (reduce) { setShown(value); return; }
    const controls = animate(0, value, {
      duration: 1.1,
      ease: [0.16, 1, 0.3, 1],
      onUpdate: (v) => setShown(Math.round(v)),
    });
    return () => controls.stop();
  }, [inView, reduce, value]);

  return (
    <span ref={ref} className={className} aria-label={`${value}${suffix}`}>
      <span aria-hidden>{shown}{suffix}</span>
    </span>
  );
}
```

```python
"""Recorta o fundo da foto e exporta os tamanhos usados no site.

Uso (fora do build, com um venv que tenha rembg e Pillow):
    python -I scripts/prepare-photo.py <foto-original.jpg> src/assets/me

O original nunca entra no repositório; só os recortes otimizados.
"""
import sys
from pathlib import Path

from PIL import Image
from rembg import new_session, remove


def main(src: str, out_dir: str) -> None:
    out = Path(out_dir)
    out.mkdir(parents=True, exist_ok=True)
    cut = remove(Image.open(src), session=new_session("birefnet-portrait")).convert("RGBA")

    bbox = cut.getchannel("A").point(lambda a: 255 if a > 8 else 0).getbbox()
    cut = cut.crop(bbox)

    h = 1200
    big = cut.resize((round(cut.width * h / cut.height), h), Image.LANCZOS)
    big.save(out / "leo-1200.webp", "WEBP", quality=86, method=6)
    big.save(out / "leo-1200.png", optimize=True)

    # Miniatura do cartão "Vamos conversar": rosto, quadrada.
    face = cut.crop((int(cut.width * 0.18), 0, int(cut.width * 0.82), int(cut.width * 0.64)))
    face.resize((160, 160), Image.LANCZOS).save(out / "leo-thumb.webp", "WEBP", quality=86)

    print(big.size, "->", out)


if __name__ == "__main__":
    main(sys.argv[1], sys.argv[2])
```

```tsx
import { motion, useMotionValue, useReducedMotion, useSpring, useTransform } from 'motion/react';
import type { PointerEvent } from 'react';

interface PolaroidProps { image: string; alt: string; title: string; tag: string; href: string }

/** Cartão de foto instantânea que inclina seguindo o cursor (mola). */
export default function Polaroid({ image, alt, title, tag, href }: PolaroidProps) {
  const reduce = useReducedMotion();
  const mx = useMotionValue(0.5);
  const my = useMotionValue(0.5);
  const rotateY = useSpring(useTransform(mx, [0, 1], [-10, 10]), { stiffness: 200, damping: 18 });
  const rotateX = useSpring(useTransform(my, [0, 1], [8, -8]), { stiffness: 200, damping: 18 });

  function onMove(e: PointerEvent<HTMLAnchorElement>) {
    if (reduce || e.pointerType !== 'mouse') return;
    const r = e.currentTarget.getBoundingClientRect();
    mx.set((e.clientX - r.left) / r.width);
    my.set((e.clientY - r.top) / r.height);
  }

  return (
    <motion.a
      href={href}
      target="_blank"
      rel="noreferrer"
      onPointerMove={onMove}
      onPointerLeave={() => { mx.set(0.5); my.set(0.5); }}
      style={{ rotateX, rotateY, rotate: 2, transformPerspective: 800 }}
      className="block w-[200px] bg-white p-2 text-ink shadow-[0_14px_30px_rgba(0,0,0,0.25)]"
```
